🎯 1. Что должно получиться

У Ани две привычки: «Растяжка» и «Чтение», и «Чтение» уже отмечено. Она сообщает боту свой город, а потом просит утреннее сообщение:

  Аня: /city Казань
  Бот: Город сохранён: Казань
       [состояние: None]
  Аня: /morning
  Бот: Доброе утро! Казань: +3, ясно.

       Ваши привычки на сегодня:
       ⬜ Растяжка
       ✅ Чтение

       Отметить — на экране /today
       [состояние: None]

Город бот запомнил: в следующий раз /morning его не спросит. Галочки — те же, что на экране /today.

Если сервис погоды не ответил, первая строка меняется, а остальное сообщение остаётся:

Доброе утро! Погоду узнать не удалось.

Ваши привычки на сегодня:
⬜ Растяжка
✅ Чтение

Отметить — на экране /today
📌 Как читать переписку:
Как в модулях 6 и 7: строки в квадратных скобках — состояние FSM, в чате их не видно. Проверки этого урока мы проводили на стенде: настоящий Dispatcher и ваши роутеры, но без серверов Telegram. Ответы Open-Meteo — настоящие, записанные 1 ноября 2026 года. У вас погода будет другой.

Почему «Казань: +3», а не «В Казани +3»

«В Казани» звучит естественнее. Но название города пишет пользователь, а склонять его бот не умеет: из «Нижний Новгород» не получить «в Нижнем Новгороде» простой заменой окончания. Поэтому город стоит в начальной форме и отделён двоеточием — так он верен для любого названия.

🌤 2. weather.py: собираем погоду в одном файле

В http_lab погода разложена по двум файлам: weather_parse.py разбирает ответы, weather_net.py отправляет запросы. В проекте это будет один модуль weather.py — как database.py отвечает за базу, так он будет отвечать за погоду.

Переносим не всё один в один. В 8.3 get_weather(city) делала два запроса подряд: по названию — координаты, по координатам — погоду. В проекте эти шаги разойдутся во времени. Город пользователь присылает один раз, командой /city, а погоду бот спрашивает каждое утро. Поэтому функций две:

find_city("Казань")                 →  ("Казань", 55.78874, 49.12214)  или None
get_weather(55.78874, 49.12214)     →  "+3, ясно"                       или None

✍️ Сначала сам

Создайте в папке проекта, рядом с database.py, файл weather.py. Что в нём должно быть:

  • импорты logging и aiohttp, константы GEO_URL, FORECAST_URL, TIMEOUT и logger — из weather_net.py;
  • словарь WEATHER_CODES и ваша describe_weather() — из weather_parse.py, без изменений;
  • get_json() из weather_net.py, без изменений;
  • parse_city() из 8.2 с одним изменением: возвращать не только координаты, но и название места — (name, latitude, longitude);
  • find_city(city): запрос к геокодеру, отказ сервиса — None и запись в лог, иначе результат parse_city();
  • get_weather(latitude, longitude): запрос прогноза, отказ — None и запись в лог, иначе результат describe_weather().

Вопрос до спойлера: зачем parse_city() теперь возвращает название, если пользователь и так его написал?

Показать решение 
"""Погода от Open-Meteo: запросы и разбор ответов."""
import logging

import aiohttp

GEO_URL = "https://geocoding-api.open-meteo.com/v1/search"
FORECAST_URL = "https://api.open-meteo.com/v1/forecast"
TIMEOUT = aiohttp.ClientTimeout(total=5)

logger = logging.getLogger(__name__)

WEATHER_CODES = {
    0: "ясно",
    1: "в основном ясно",
    2: "переменная облачность",
    3: "пасмурно",
    45: "туман",
    48: "туман с изморозью",
    51: "слабая морось",
    53: "морось",
    55: "сильная морось",
    61: "слабый дождь",
    63: "дождь",
    65: "сильный дождь",
    71: "слабый снег",
    73: "снег",
    75: "сильный снег",
    80: "ливень",
    95: "гроза",
}


async def get_json(url: str, params: dict) -> dict:
    async with aiohttp.ClientSession(timeout=TIMEOUT) as session:
        async with session.get(url, params=params) as response:
            response.raise_for_status()
            return await response.json()


def parse_city(data: dict) -> tuple[str, float, float] | None:
    results = data.get("results")
    if not results:
        return None

    place = results[0]
    name = place.get("name")
    latitude = place.get("latitude")
    longitude = place.get("longitude")
    if name is None or latitude is None or longitude is None:
        return None
    return name, latitude, longitude


def describe_weather(data: dict) -> str | None:
    current = data.get("current")
    if not current:
        return None

    temperature = current.get("temperature_2m")
    code = current.get("weather_code")
    if temperature is None or code is None:
        return None

    degrees = round(temperature)
    if degrees == 0:
        sign = "0"
    else:
        sign = f"{degrees:+d}"
    text = WEATHER_CODES.get(code, "без описания")
    return f"{sign}, {text}"


async def find_city(city: str) -> tuple[str, float, float] | None:
    params = {"name": city, "count": 1, "language": "ru"}
    try:
        data = await get_json(GEO_URL, params)
    except (aiohttp.ClientError, TimeoutError) as error:
        name = type(error).__name__
        logger.warning("Геокодер недоступен для %s: %s", city, name)
        return None

    place = parse_city(data)
    if place is None:
        logger.info("Город не найден: %s", city)
    return place


async def get_weather(latitude: float, longitude: float) -> str | None:
    params = {
        "latitude": latitude,
        "longitude": longitude,
        "current": "temperature_2m,weather_code",
    }
    try:
        data = await get_json(FORECAST_URL, params)
    except (aiohttp.ClientError, TimeoutError) as error:
        name = type(error).__name__
        logger.warning("Погода недоступна: %s", name)
        return None

    text = describe_weather(data)
    if text is None:
        logger.warning("Неожиданный ответ прогноза")
    return text

Что здесь новое

Почти ничего: каждая часть файла уже работала в лаборатории. Разберём отличия.

parse_city() возвращает название. Ответ на вопрос до спойлера: пользователь напишет город как получится, а геокодер вернёт его так, как он называется. В /city бот покажет и сохранит именно это название. Проверку name is None добавили по правилу урока 8.2: в чужом ответе может не оказаться любого ключа.

find_city() и get_weather() устроены одинаково. Запрос в try, отказ сервиса — запись в лог и None, иначе — разбор ответа. Это форма get_weather() из урока 8.3, только каждая функция делает один запрос.

Отказ и «не нашлось» дают одно и то же None. Для геокодера «нет такого города» и «сервис не ответил» — разные события. Но пользователю мы скажем одно: проверьте название или попробуйте позже. А различать их будем по логу: info для ненайденного города, warning для отказа.

Лог в терминале бота. В bot.py логирование не настраивается. Тогда Python всё равно печатает в терминал записи уровня warning и выше, только без названия уровня. Записи info не печатаются. Для отказов сервиса этого достаточно: их мы увидим.

Вопрос: Почему в проекте get_weather() принимает координаты, а не название города, как в уроке 8.3?

Подсказка при неверном ответе:

🗄 3. Таблица users: город и координаты

Боту нужно помнить город каждого пользователя. Привычки лежат в базе, и город туда же: новая таблица, по строке на пользователя.

Что хранить? Название — чтобы показать его в сообщении. И координаты — чтобы не искать город каждое утро. Если хранить только название, утром боту пришлось бы снова спрашивать геокодер: два запроса вместо одного, и два шанса, что сервис не ответит.

Шаг 1. Таблица

Добавьте таблицу в строку CREATE_TABLES в database.py, после таблицы marks:

CREATE TABLE IF NOT EXISTS users (
    user_id   INTEGER PRIMARY KEY,
    city      TEXT    NOT NULL,
    latitude  REAL    NOT NULL,
    longitude REAL    NOT NULL
);

Новое здесь — два момента.

user_id INTEGER PRIMARY KEY. У таблиц habits и marks ключ — id, его выдаёт база. Здесь ключом служит сам номер пользователя: у одного пользователя один город. Две строки с одним user_id база не допустит.

REAL — дробное число. Координаты — дроби: 55.78874.

Шаг 2. Таблица в уже существующей базе

У вас есть habits.db после урока 7.4, с привычками и отметками. Удалять его не нужно. Посмотрим, какие в нём таблицы. Остановите бота и выполните в папке проекта:

python -c "import sqlite3; db = sqlite3.connect('habits.db'); print(db.execute('SELECT name FROM sqlite_master WHERE type = ?', ('table',)).fetchall())"

sqlite_master — служебная таблица, в которой SQLite хранит список всех таблиц. Значение подставлено через ?, как в уроке 7.2. До изменений:

[('habits',), ('sqlite_sequence',), ('marks',)]

sqlite_sequence SQLite завёл сам: в ней он помнит последний выданный номер для AUTOINCREMENT. Таблицы users нет.

Запустите бота с новым CREATE_TABLES, остановите и выполните ту же команду:

[('habits',), ('sqlite_sequence',), ('marks',), ('users',)]

users появилась. Её создала init_db() при запуске, как в уроке 7.4. CREATE TABLE IF NOT EXISTS пропустил habits и marks — они уже были — и создал недостающую. А привычки Ани на месте:

  Аня: /today
  Бот: Мои привычки на сегодня
       текущая дата

       Выполнено: 1 из 2

       [ ⬜ Растяжка ]
       [ ✅ Чтение ]
       [состояние: None]

Вот зачем в уроке 7.2 мы писали IF NOT EXISTS. Схема растёт вместе с ботом, а данные пользователей остаются.

✍️ Сначала сам: set_city() и get_city()

Добавьте в конец database.py две функции:

  • set_city(user_id, city, latitude, longitude) — записать город пользователя;
  • get_city(user_id) — вернуть словарь {"city": ..., "latitude": ..., "longitude": ...} или None, если города нет. Устроена как get_habit().

Вопрос до спойлера: что случится, если пользователь отправит /city второй раз, а set_city() делает обычный INSERT?

Показать решение

Сначала — set_city() с обычным INSERT, как add_habit():

async def set_city(
    user_id: int, city: str, latitude: float, longitude: float
) -> None:
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute(
            "INSERT INTO users (user_id, city, latitude, longitude) "
            "VALUES (?, ?, ?, ?)",
            (user_id, city, latitude, longitude),
        )
        await db.commit()

Проверим: запишем Ане Казань дважды. Первый раз:

python -c "import asyncio, database; asyncio.run(database.set_city(111, 'Казань', 55.78874, 49.12214)); print(asyncio.run(database.get_city(111)))"
{'city': 'Казань', 'latitude': 55.78874, 'longitude': 49.12214}

Второй раз та же команда:

Traceback (most recent call last):
  ...
  File "database.py", line 171, in set_city
    await db.execute(
  ...
sqlite3.IntegrityError: UNIQUE constraint failed: users.user_id

Строка с user_id = 111 уже есть, а user_id — ключ. База отказалась вставлять вторую. В боте это значит: сменить город нельзя.

Нужно «вставь, а если такой пользователь уже есть — обнови». В SQLite это пишется так:

INSERT INTO users (user_id, city, latitude, longitude)
VALUES (?, ?, ?, ?)
ON CONFLICT (user_id) DO UPDATE SET
    city = excluded.city,
    latitude = excluded.latitude,
    longitude = excluded.longitude
  • ON CONFLICT (user_id) — что делать, если строка с таким user_id уже есть;
  • DO UPDATE SET — обновить её вместо вставки;
  • excluded.city — значение, которое мы пытались вставить. excluded — «исключённая» строка, та, что не прошла из-за конфликта.

Итоговая set_city():

async def set_city(
    user_id: int, city: str, latitude: float, longitude: float
) -> None:
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute(
            "INSERT INTO users (user_id, city, latitude, longitude) "
            "VALUES (?, ?, ?, ?) "
            "ON CONFLICT (user_id) DO UPDATE SET "
            "city = excluded.city, "
            "latitude = excluded.latitude, "
            "longitude = excluded.longitude",
            (user_id, city, latitude, longitude),
        )
        await db.commit()

И get_city():

async def get_city(user_id: int) -> dict | None:
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        cursor = await db.execute(
            "SELECT city, latitude, longitude FROM users WHERE user_id = ?",
            (user_id,),
        )
        row = await cursor.fetchone()
    if row is None:
        return None
    return dict(row)

Что здесь новое

Ответ на вопрос до спойлера вы видели: IntegrityError. Обычный INSERT годится, когда каждая запись новая: привычка, отметка. Для настройки, которая у пользователя одна, нужен INSERT ... ON CONFLICT DO UPDATE. Такую запись часто называют upsert: от update и insert.

get_city() ничего нового не содержит: row_factory, fetchone() и None для пустого результата — как в get_habit() из урока 7.2.

🏙 4. Команда /city Казань

У команды /city есть то, чего не было ни у одной команды бота: продолжение. Город пишут в той же строке, после пробела: /city Казань. Как хэндлеру получить эту часть?

CommandObject: команда по частям

Фильтр Command("city") не только проверяет, что сообщение начинается с /city. Он разбирает его на части и кладёт результат в объект CommandObject. Чтобы получить его в хэндлере, добавьте параметр с именем command:

async def cmd_city(message: Message, command: CommandObject) -> None:

Так же aiogram передаёт state в хэндлеры диалога (модуль 6): видит параметр с известным именем и подставляет значение. И правило то же: параметр должен называться именно command. Назовёте иначе — бот не ответит, а в терминале будет TypeError: ... missing 1 required positional argument, как с ctx вместо state.

Текст после команды лежит в command.args. Мы проверили, что там оказывается:

/city                         →  command.args = None
/city Казань                  →  command.args = "Казань"
/city   Нижний Новгород       →  command.args = "Нижний Новгород"
/city Казань   (с пробелами)  →  command.args = "Казань   "

Без города — None, а не пустая строка. Пробелы слева aiogram убирает, справа оставляет. Поэтому дальше: проверка на пустоту и .strip().

async def cmd_city(message: Message, command: CommandObject) -> None:

☀️ 5. build_morning_text(): утреннее сообщение

Город есть, погода есть, привычки есть. Соберём из них текст. Сделаем это отдельной функцией, а не внутри хэндлера, и вот почему.

Сейчас сообщение будет приходить в ответ на команду. А в модуле 9 бот начнёт присылать его сам, по времени. Отправлять будет цикл напоминаний — и у него нет message: никто ничего не писал. Есть только номер пользователя и сегодняшняя дата. Значит, функция должна обходиться ими:

build_morning_text(user_id, day)  →  готовый текст сообщения

Так же в уроке 5.3 мы вынесли build_today_screen(): экран собирает одна функция, а вызывают её разные хэндлеры.

Тексты

Добавьте в конец texts.py:

MORNING = (
    "Доброе утро! {weather}\n\n"
    "<b>Ваши привычки на сегодня:</b>\n"
    "{habits}\n\n"
    "Отметить — на экране /today"

✍️ Сначала сам

Напишите в handlers/habits.py, после cmd_city(), функцию build_morning_text(user_id, day):

  • получить город через database.get_city(); нет города — строка погоды NO_CITY;
  • есть — спросить weather.get_weather() по координатам; None — NO_WEATHER, иначе WEATHER_LINE;
  • получить привычки и отметки: get_habits() и get_done_ids() из урока 7.4;
  • нет привычек — вместо списка EMPTY_TODAY; есть — по строке на привычку: ✅ или ⬜ и название;
  • вернуть MORNING с подставленными строками.

Показать решение 
async def build_morning_text(user_id: int, day: date) -> str:
    place = await database.get_city(user_id)
    if place is None:
        weather_line = texts.NO_CITY
    else:
        forecast = await weather.get_weather(
            place["latitude"], place["longitude"]
        )
        if forecast is None:
            weather_line = texts.NO_WEATHER
        else:
            weather_line = texts.WEATHER_LINE.format(
                city=html.quote(place["city"]),
                weather=forecast,
            )

    habits = await database.get_habits(user_id)
    done_ids = await database.get_done_ids(user_id, day)
    if not habits:
        habits_text = texts.EMPTY_TODAY
    else:
        lines = []
        for habit in habits:
            mark = "✅" if habit["id"] in done_ids else "⬜"
            lines.append(f"{mark} {html.quote(habit['name'])}")
        habits_text = "\n".join(lines)

    return texts.MORNING.format(weather=weather_line, habits=habits_text)

Что здесь новое

Ни одной новой функции базы. get_habits() и get_done_ids() — те же, что собирают экран /today в уроке 7.4. Поэтому галочки в утреннем сообщении всегда совпадают с экраном: данные одни. Отметка ✅ ставится тем же выражением, что на кнопках в keyboards.py.

Три исхода погоды — три строки. Как в наброске morning_demo.py из урока 8.3: None не роняет сообщение, а меняет одну строку. Привычки собираются в любом случае.

Ответ на вопрос до спойлера. weather — имя модуля, который мы импортировали. Если в функции где-то есть присваивание weather = ..., Python считает weather локальной переменной во всей функции. Тогда строка weather = await weather.get_weather(...) обращается к ещё не созданной переменной. Мы проверили:

UnboundLocalError: cannot access local variable 'weather' where it is not associated with a value

Поэтому ответ лежит в forecast, а строка для сообщения — в weather_line.

html.quote() — для названия города и привычек: оба пришли со стороны.

"""Погода от Open-Meteo: запросы и разбор ответов."""
import logging

import aiohttp

GEO_URL = "https://geocoding-api.open-meteo.com/v1/search"
FORECAST_URL = "https://api.open-meteo.com/v1/forecast"
TIMEOUT = aiohttp.ClientTimeout(total=5)

logger = logging.getLogger(__name__)

WEATHER_CODES = {
    0: "ясно",
    1: "в основном ясно",
    2: "переменная облачность",
    3: "пасмурно",
    45: "туман",
    48: "туман с изморозью",
    51: "слабая морось",
    53: "морось",
    55: "сильная морось",
    61: "слабый дождь",
    63: "дождь",
    65: "сильный дождь",
    71: "слабый снег",
    73: "снег",
    75: "сильный снег",
    80: "ливень",
    95: "гроза",
}


async def get_json(url: str, params: dict) -> dict:
    async with aiohttp.ClientSession(timeout=TIMEOUT) as session:
        async with session.get(url, params=params) as response:
            response.raise_for_status()
            return await response.json()


def parse_city(data: dict) -> tuple[str, float, float] | None:
    results = data.get("results")
    if not results:
        return None

    place = results[0]
    name = place.get("name")
    latitude = place.get("latitude")
    longitude = place.get("longitude")
    if name is None or latitude is None or longitude is None:
        return None
    return name, latitude, longitude


def describe_weather(data: dict) -> str | None:
    current = data.get("current")
    if not current:
        return None

    temperature = current.get("temperature_2m")
    code = current.get("weather_code")
    if temperature is None or code is None:
        return None

    degrees = round(temperature)
    if degrees == 0:
        sign = "0"
    else:
        sign = f"{degrees:+d}"
    text = WEATHER_CODES.get(code, "без описания")
    return f"{sign}, {text}"


async def find_city(city: str) -> tuple[str, float, float] | None:
    params = {"name": city, "count": 1, "language": "ru"}
    try:
        data = await get_json(GEO_URL, params)
    except (aiohttp.ClientError, TimeoutError) as error:
        name = type(error).__name__
        logger.warning("Геокодер недоступен для %s: %s", city, name)
        return None

    place = parse_city(data)
    if place is None:
        logger.info("Город не найден: %s", city)
    return place


async def get_weather(latitude: float, longitude: float) -> str | None:
    params = {
        "latitude": latitude,
        "longitude": longitude,
        "current": "temperature_2m,weather_code",
    }
    try:
        data = await get_json(FORECAST_URL, params)
    except (aiohttp.ClientError, TimeoutError) as error:
        name = type(error).__name__
        logger.warning("Погода недоступна: %s", name)
        return None

    text = describe_weather(data)
    if text is None:
        logger.warning("Неожиданный ответ прогноза")
    return text

async def build_today_screen(
    user_id: int,
    day: date,
    page: int,
) -> tuple[str, InlineKeyboardMarkup | None]:

@router.message(Command("city"))
async def cmd_city(message: Message, command: CommandObject) -> None:
    if not command.args:
        await message.answer(texts.CITY_USAGE)
        return

    query = command.args.strip()
    place = await weather.find_city(query)
    if place is None:
        await message.answer(
            texts.CITY_NOT_FOUND.format(city=html.quote(query))
        )
        return

    await database.set_city(message.from_user.id, city, latitude, longitude)
    await message.answer(texts.CITY_SAVED.format(city=html.quote(city)))

async def build_morning_text(user_id: int, day: date) -> str:
    place = await database.get_city(user_id)
    if place is None:
        weather_line = texts.NO_CITY
    else:
        forecast = await weather.get_weather(
            place["latitude"], place["longitude"]
        )
        if forecast is None:
            weather_line = texts.NO_WEATHER
        else:
            weather_line = texts.WEATHER_LINE.format(
                city=html.quote(place["city"]),
                weather=forecast,
            )

    habits = await database.get_habits(user_id)
    done_ids = await database.get_done_ids(user_id, day)
    if not habits:
        habits_text = texts.EMPTY_TODAY
    else:
        lines = []
        for habit in habits:
            mark = "✅" if habit["id"] in done_ids else "⬜"
            lines.append(f"{mark} {html.quote(habit['name'])}")
        habits_text = "\n".join(lines)

    return texts.MORNING.format(weather=weather_line, habits=habits_text)

            texts.CB_MESSAGE_UNAVAILABLE,
            show_alert=True,
        )
        return

    payload = callback.data.removeprefix("toggle_")
    parts = payload.split("_")

    if (
        len(parts) != 2
        or not parts[0].isdigit()
        or not parts[1].isdigit()

        await callback.answer(
            texts.CB_OLD_BUTTON,

        return

    habit_id = int(parts[0])
    page = int(parts[1])

    habit = await database.get_habit(callback.from_user.id, habit_id)

    if habit is None:
        await callback.answer(

            show_alert=True,

    day = date.today()
    marked = await database.toggle_mark(habit_id, day)

    await edit_today_message(
        callback.message,
        user_id=callback.from_user.id,
        day=day,
        page=page,

        await callback.answer(texts.CB_MARKED)

        await callback.answer(texts.CB_UNDONE)

"""Хранилище ПривычкаБота в базе SQLite."""
from datetime import date

import aiosqlite

DB_PATH = "habits.db"

CREATE_TABLES = """
CREATE TABLE IF NOT EXISTS habits (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    user_id     INTEGER NOT NULL,
    name        TEXT    NOT NULL,
    frequency   INTEGER NOT NULL,
    remind_time TEXT    NOT NULL,
    created_at  TEXT    NOT NULL DEFAULT (datetime('now'))
);

CREATE TABLE IF NOT EXISTS marks (
    id          INTEGER PRIMARY KEY AUTOINCREMENT,
    habit_id    INTEGER NOT NULL,
    marked_date TEXT    NOT NULL,
    UNIQUE (habit_id, marked_date)
);

CREATE TABLE IF NOT EXISTS users (
    user_id   INTEGER PRIMARY KEY,
    city      TEXT    NOT NULL,
    latitude  REAL    NOT NULL,
    longitude REAL    NOT NULL
);
"""


async def init_db() -> None:
    async with aiosqlite.connect(DB_PATH) as db:
        await db.executescript(CREATE_TABLES)
        await db.commit()


async def add_habit(
    user_id: int, name: str, frequency: int, remind_time: str
) -> dict:
    async with aiosqlite.connect(DB_PATH) as db:
        cursor = await db.execute(
            "INSERT INTO habits (user_id, name, frequency, remind_time) "
            "VALUES (?, ?, ?, ?)",
            (user_id, name, frequency, remind_time),
        )
        await db.commit()
    return {
        "id": cursor.lastrowid,
        "name": name,
        "frequency": frequency,
        "remind_time": remind_time,
    }


async def get_habits(user_id: int) -> list[dict]:
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        cursor = await db.execute(
            "SELECT id, name, frequency, remind_time FROM habits "
            "WHERE user_id = ? ORDER BY id",
            (user_id,),
        )
        rows = await cursor.fetchall()
    return [dict(row) for row in rows]


async def get_habit(user_id: int, habit_id: int) -> dict | None:
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        cursor = await db.execute(
            "SELECT id, name, frequency, remind_time FROM habits "
            "WHERE id = ? AND user_id = ?",
            (habit_id, user_id),
        )
        row = await cursor.fetchone()
    if row is None:
        return None
    return dict(row)


async def toggle_mark(habit_id: int, day: date) -> bool:
    async with aiosqlite.connect(DB_PATH) as db:
        cursor = await db.execute(
            "DELETE FROM marks WHERE habit_id = ? AND marked_date = ?",
            (habit_id, day.isoformat()),
        )
        if cursor.rowcount > 0:
            await db.commit()
            return False

        await db.execute(
            "INSERT INTO marks (habit_id, marked_date) VALUES (?, ?)",
            (habit_id, day.isoformat()),
        )
        await db.commit()
        return True


async def count_done(user_id: int, day: date) -> int:
    async with aiosqlite.connect(DB_PATH) as db:
        cursor = await db.execute(
            "SELECT COUNT(*) "
            "FROM marks m "
            "JOIN habits h ON h.id = m.habit_id "
            "WHERE h.user_id = ? AND m.marked_date = ?",
            (user_id, day.isoformat()),
        )
        row = await cursor.fetchone()
    return row[0]


async def get_stats(user_id: int, since: date, until: date) -> list[dict]:
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        cursor = await db.execute(
            "SELECT h.id, h.name, COUNT(m.id) AS done "
            "FROM habits h "
            "LEFT JOIN marks m ON m.habit_id = h.id "
            "AND m.marked_date >= ? AND m.marked_date <= ? "
            "WHERE h.user_id = ? "
            "GROUP BY h.id, h.name "
            "ORDER BY done DESC, h.id",
            (since.isoformat(), until.isoformat(), user_id),
        )
        rows = await cursor.fetchall()
    return [dict(row) for row in rows]


async def get_mark_days(user_id: int, habit_id: int, today: date) -> list[date]:
    async with aiosqlite.connect(DB_PATH) as db:
        cursor = await db.execute(
            "SELECT m.marked_date "
            "FROM marks m "
            "JOIN habits h ON h.id = m.habit_id "
            "WHERE m.habit_id = ? AND h.user_id = ? AND m.marked_date <= ? "
            "ORDER BY m.marked_date DESC",
            (habit_id, user_id, today.isoformat()),
        )
        rows = await cursor.fetchall()

    days = []
    for row in rows:
        days.append(date.fromisoformat(row[0]))
    return days


async def get_done_ids(user_id: int, day: date) -> set[int]:
    async with aiosqlite.connect(DB_PATH) as db:
        cursor = await db.execute(
            "SELECT m.habit_id "
            "FROM marks m "
            "JOIN habits h ON h.id = m.habit_id "
            "WHERE h.user_id = ? AND m.marked_date = ?",
            (user_id, day.isoformat()),
        )
        rows = await cursor.fetchall()

    done_ids = set()
    for row in rows:
        done_ids.add(row[0])
    return done_ids


async def set_city(
    user_id: int, city: str, latitude: float, longitude: float
) -> None:
    async with aiosqlite.connect(DB_PATH) as db:
        await db.execute(
            "INSERT INTO users (user_id, city, latitude, longitude) "
            "VALUES (?, ?, ?, ?) "
            "ON CONFLICT (user_id) DO UPDATE SET "
            "city = excluded.city, "
            "latitude = excluded.latitude, "
            "longitude = excluded.longitude",
            (user_id, city, latitude, longitude),
        )
        await db.commit()


async def get_city(user_id: int) -> dict | None:
    async with aiosqlite.connect(DB_PATH) as db:
        db.row_factory = aiosqlite.Row
        cursor = await db.execute(
            "SELECT city, latitude, longitude FROM users WHERE user_id = ?",
            (user_id,),
        )
        row = await cursor.fetchone()
    if row is None:
        return None
    return dict(row)
